# SentenceMaker

Build a Sentence Maker Project.

## Source Code

```js
let adjective = "happy",
  noun = "monkey",
  verb = "jumped",
  place = "big villa",
  adjective2 = "big",
  noun2 = "bananas";
let firstStory =
  "Once upon a time, there was a(n) " +
  adjective +
  " " +
  noun +
  " who loved to eat " +
  noun2 +
  ". The " +
  noun +
  " lived in a " +
  place +
  " and had " +
  adjective2 +
  " nostrils that blew fire when it was " +
  verb +
  ".";

console.log("First story: " + firstStory);

((adjective = "angry"),
  (noun = "lion"),
  (verb = "bawling"),
  (place = "the jungle"),
  (adjective2 = "strong"),
  (noun2 = "meat"));

let secondStory =
  "Once upon a time, there was a(n) " +
  adjective +
  " " +
  noun +
  " who loved to eat " +
  noun2 +
  ". The " +
  noun +
  " lived in a " +
  place +
  " and had " +
  adjective2 +
  " nostrils that blew fire when it was " +
  verb +
  ".";

console.log("Second story: " + secondStory);
```
